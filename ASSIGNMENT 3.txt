5. Find hottest and coolest month from ncdc data.

CREATE TABLE ncdc_staging_formonth(
yr SMALLINT,
mont SMALLINT,
dy SMALLINT,
temp SMALLINT,
quality TINYINT
)
ROW FORMAT SERDE 'org.apache.hadoop.hive.serde2.RegexSerDe'
WITH SERDEPROPERTIES('input.regex'='^.{15}([0-9]{4})([0-9]{2})([0-9]{2}).{64}([-/+][0-9]{4})([0-9]).*$');

LOAD DATA LOCAL
INPATH '/home/sunbeam/Desktop/SUNBEAM/Big/BigData/data/ncdc/*'
INTO TABLE ncdc_staging_formonth;


SELECT * FROM ncdc_staging_formonth limit 10;

WITH cte assh
(SELECT MAX(temp) as mx,MIN(temp) as mn FROM ncdc_staging_formonth
WHERE quality IN(0,1,2,4,5,9) AND temp !=9999)
SELECT n.mont,n.temp FROM ncdc_staging_formonth as n 
JOIN cte AS c
WHERE n.temp IN (c.mx,c.mn);
